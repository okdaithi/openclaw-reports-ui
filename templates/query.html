Example Queries

List all reports: 
SELECT report_date, report_type, country, risk_score 
FROM reports 
ORDER BY report_date DESC 
LIMIT 20;
High-risk reports: 
SELECT report_date, country, risk_score, volatility_score 
FROM reports 
WHERE risk_score > 70 
ORDER BY risk_score DESC;
Entity mentions: 
SELECT e.name, e.type, COUNT(*) as mention_count 
FROM entities e 
JOIN entity_mentions em ON e.id = em.entity_id 
GROUP BY e.name, e.type 
ORDER BY mention_count DESC 
LIMIT 20;
Country risk trend (view): 
SELECT * FROM v_country_risk_trend ORDER BY avg_risk DESC;
Recent signals: 
SELECT category, subcategory, COUNT(*) as count 
FROM signals 
GROUP BY category, subcategory 
ORDER BY count DESC;
